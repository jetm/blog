{{- /* Custom Render Hook for Markdown Images Processes inline images with responsive srcset, configurable formats/quality, and lazy loading. Delegates to process-image partial for processable images. Preserves user-specified markdown attributes (class, id, width, height, loading, etc.). */ -}} {{- $u := urls.Parse .Destination -}} {{- $src := $u.String -}} {{- $alt := .Text -}} {{- $title := .Title -}} {{- $userAttrs := .Attributes | default dict -}} {{- $image := false -}} {{- /* Resolve image resource */ -}} {{- if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- with or (.PageInner.Resources.Get $path) (resources.Get $path) -}} {{- $image = . -}} {{- $src = .RelPermalink -}} {{- with $u.RawQuery -}} {{- $src = printf "%s?%s" $src . -}} {{- end -}} {{- with $u.Fragment -}} {{- $src = printf "%s#%s" $src . -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Define processable formats */ -}} {{- $processableFormats := (slice "jpg" "jpeg" "png" "tif" "bmp") -}} {{- if hugo.IsExtended -}} {{- $processableFormats = $processableFormats | append "webp" -}} {{- end -}} {{- $prod := (hugo.IsProduction | or (eq site.Params.env "production")) -}} {{- if and $image (in $processableFormats $image.MediaType.SubType) $prod -}} {{- /* Delegate to process-image partial for processable images */ -}} {{- partial "process-image.html" (dict "image" $image "alt" $alt "title" $title "attributes" $userAttrs "asFigure" (ne $title "") ) -}} {{- else -}} {{- /* Non-processable image or external URL - render with merged attributes */ -}} {{- /* Build base attributes */ -}} {{- $baseAttrs := dict "src" $src "alt" $alt "loading" "lazy" "decoding" "async" -}} {{- /* Add width/height if we have the image resource (skip for GIF to preserve intrinsic size) */ -}} {{- if and $image (ne $image.MediaType.SubType "gif") -}} {{- $baseAttrs = merge (dict "width" (printf "%d" $image.Width) "height" (printf "%d" $image.Height) ) $baseAttrs -}} {{- end -}} {{- if $title -}} {{- $baseAttrs = merge (dict "title" ($title | transform.HTMLEscape)) $baseAttrs -}} {{- end -}} {{- /* Merge user attributes - they override base defaults */ -}} {{- $attrs := merge $userAttrs $baseAttrs -}} {{- if $title -}} {{- else -}} {{- end -}} {{- end -}}
