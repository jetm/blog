{{- /* Process Image Partial Generates responsive images with configurable formats, quality, and srcset attributes. Parameters (passed as dict): - image: Hugo image resource (required) - alt: Alt text for accessibility (required) - title: Optional title attribute - loading: "lazy" or "eager" (default: "lazy") - class: Optional CSS class names - attributes: Additional HTML attributes dict to merge (user-specified attrs take precedence) - sizes: Responsive sizes array (default from site params) - asFigure: Whether to wrap in figure with caption (default: false) */ -}} {{- $image := .image -}} {{- $alt := .alt | default "" -}} {{- $title := .title | default "" -}} {{- $loading := .loading | default "lazy" -}} {{- $class := .class | default "" -}} {{- $userAttrs := .attributes | default dict -}} {{- $responsiveSizes := .sizes | default (site.Params.imageProcessing.responsiveSizes | default (slice 360 480 720 1080 1500)) -}} {{- $asFigure := .asFigure | default false -}} {{- /* Get configured formats and quality */ -}} {{- $configFormats := site.Params.imageProcessing.formats | default (slice "webp") -}} {{- $quality := site.Params.imageProcessing.quality | default 75 -}} {{- $prod := (hugo.IsProduction | or (eq site.Params.env "production")) -}} {{- /* Define processable formats */ -}} {{- $processableFormats := (slice "jpg" "jpeg" "png" "tif" "bmp") -}} {{- if hugo.IsExtended -}} {{- $processableFormats = $processableFormats | append "webp" -}} {{- end -}} {{- $isProcessable := in $processableFormats $image.MediaType.SubType -}} {{- /* Build base attributes (defaults that user can override) */ -}} {{- $baseAttrs := dict "width" (printf "%d" $image.Width) "height" (printf "%d" $image.Height) "alt" $alt "loading" $loading "decoding" "async" -}} {{- if $title -}} {{- $baseAttrs = merge (dict "title" ($title | transform.HTMLEscape)) $baseAttrs -}} {{- end -}} {{- if $class -}} {{- $baseAttrs = merge (dict "class" $class) $baseAttrs -}} {{- end -}} {{- /* Merge user attributes - they override base defaults */ -}} {{- $attrs := merge $userAttrs $baseAttrs -}} {{- /* Build srcset and set computed attributes */ -}} {{- if and $isProcessable $prod -}} {{- $srcset := slice -}} {{- /* Generate variants for each configured format */ -}} {{- range $format := $configFormats -}} {{- range $size := $responsiveSizes -}} {{- if ge $image.Width (int $size) -}} {{- $resizeSpec := printf "%dx %s q%d" (int $size) $format (int $quality) -}} {{- $resized := $image.Resize $resizeSpec -}} {{- $srcset = $srcset | append (printf "%s %dw" $resized.RelPermalink (int $size)) -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Add original size as fallback */ -}} {{- $srcset = $srcset | append (printf "%s %dw" $image.RelPermalink $image.Width) -}} {{- /* Set computed attributes - these override user attrs for src/srcset/sizes */ -}} {{- $attrs = merge (dict "src" $image.RelPermalink "srcset" (delimit $srcset ", ") "sizes" "(min-width: 768px) 720px, 100vw" ) $attrs -}} {{- else -}} {{- /* Non-processable or dev mode - just set src */ -}} {{- $attrs = merge (dict "src" $image.RelPermalink) $attrs -}} {{- end -}} {{- /* Render the image or figure */ -}} {{- if and $asFigure $title -}} {{- else -}} {{- end -}}
